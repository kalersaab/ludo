cmake_minimum_required(VERSION 3.16)

project(ludo_native LANGUAGES CXX)

add_library(ludo_native STATIC ludo.cc)

target_compile_features(ludo_native PUBLIC cxx_std_17)

target_include_directories(ludo_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

if (MSVC)
  target_compile_options(ludo_native PRIVATE /W4)
else()
  target_compile_options(ludo_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
